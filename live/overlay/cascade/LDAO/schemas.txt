establish(Alice?) :
    at Alice: not identity, +identity

acquaint(Alice?, Bob?) :
    at Alice: identity, +knows(Bob)
    at Bob: identity, +knows(Alice)

introduce(Alice?, Bob?; c) :
    at Alice: identity, knows(Bob), knows(c)
    at Bob: identity, knows(Alice), +knows(c)

found(Alice?; n) :
    at Alice: identity, not member(<Alice>.n), +member(<Alice>.n), +listed(Alice, n)

admit(Alice?, Bob?; n) :
    at Alice: identity, member(<Alice>.n), listed(Alice, n), knows(Bob), not listed(Bob, n), +listed(Bob, n)
    at Bob: identity, not member(<Alice>.n), +member(<Alice>.n)

propose(Bob?, Alice; x, n) :
    at Bob: identity, member(<Alice>.n)
    at Alice: identity, member(<Alice>.n), listed(Alice, n), listed(Bob, n), knows(Bob), not proposal(x, Bob, n), +proposal(x, Bob, n)

propose_own(Alice?; x, n) :
    at Alice: identity, member(<Alice>.n), listed(Alice, n), not proposal(x, Alice, n), +proposal(x, Alice, n)

deliver_proposal(Alice, Bob; x, o, n) :
    at Alice: identity, member(<Alice>.n), listed(Alice, n), proposal(x, o, n), listed(Bob, n), knows(o), not sent(Bob, x, n), +sent(Bob, x, n)
    at Bob: identity, member(<Alice>.n), +proposal(x, o, n)

deliver_own_proposal(Alice, Bob; x, n) :
    at Alice: identity, member(<Alice>.n), listed(Alice, n), proposal(x, Alice, n), listed(Bob, n), not sent(Bob, x, n), +sent(Bob, x, n)
    at Bob: identity, member(<Alice>.n), +proposal(x, Alice, n)

vote_for(Bob?, Alice; x, o, n) :
    at Bob: identity, member(<Alice>.n), proposal(x, o, n), not voted(x, o, n), +voted(x, o, n)
    at Alice: identity, member(<Alice>.n), listed(Alice, n), listed(Bob, n), proposal(x, o, n), not counted(x, o, Bob, n), not decided(x, o, n), +yes_vote(x, o, Bob, n), +counted(x, o, Bob, n)

vote_own_for(Alice?; x, o, n) :
    at Alice: identity, member(<Alice>.n), listed(Alice, n), proposal(x, o, n), not counted(x, o, Alice, n), not voted(x, o, n), not decided(x, o, n), +yes_vote(x, o, Alice, n), +counted(x, o, Alice, n), +voted(x, o, n)

vote_against(Bob?, Alice; x, o, n) :
    at Bob: identity, member(<Alice>.n), proposal(x, o, n), not voted(x, o, n), +voted(x, o, n)
    at Alice: identity, member(<Alice>.n), listed(Alice, n), listed(Bob, n), proposal(x, o, n), not counted(x, o, Bob, n), not decided(x, o, n), +no_vote(x, o, Bob, n), +counted(x, o, Bob, n)

vote_own_against(Alice?; x, o, n) :
    at Alice: identity, member(<Alice>.n), listed(Alice, n), proposal(x, o, n), not counted(x, o, Alice, n), not voted(x, o, n), not decided(x, o, n), +no_vote(x, o, Alice, n), +counted(x, o, Alice, n), +voted(x, o, n)

pass(Alice?; x, o, n) :
    at Alice: identity, member(<Alice>.n), listed(Alice, n), proposal(x, o, n), not decided(x, o, n), +passed_proposal(x, o, n), +decided(x, o, n)

deliver_pass(Alice, Bob; x, o, n) :
    at Alice: identity, member(<Alice>.n), listed(Alice, n), passed_proposal(x, o, n), listed(Bob, n), not told(Bob, x, o, n), +told(Bob, x, o, n)
    at Bob: identity, member(<Alice>.n), +passed_proposal(x, o, n), +decided(x, o, n)

reject(Alice?; x, o, n) :
    at Alice: identity, member(<Alice>.n), listed(Alice, n), proposal(x, o, n), not decided(x, o, n), +rejected_proposal(x, o, n), +decided(x, o, n)

deliver_reject(Alice, Bob; x, o, n) :
    at Alice: identity, member(<Alice>.n), listed(Alice, n), rejected_proposal(x, o, n), listed(Bob, n), not told(Bob, x, o, n), +told(Bob, x, o, n)
    at Bob: identity, member(<Alice>.n), +rejected_proposal(x, o, n), +decided(x, o, n)

delegate(Bob?, Alice; c, n) :
    at Bob: identity, member(<Alice>.n), knows(c), not delegating(n), +my_delegate(c, n), +delegating(n)
    at Alice: identity, member(<Alice>.n), listed(Alice, n), listed(Bob, n), listed(c, n), +delegation(Bob, c, n)

revoke(Bob?, Alice; c, n) :
    at Bob: identity, member(<Alice>.n), -my_delegate(c, n), -delegating(n)
    at Alice: identity, member(<Alice>.n), -delegation(Bob, c, n)
